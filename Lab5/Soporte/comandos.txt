# Fase B: endurecimiento (en DMZ.10 por SSH)
cp /etc/nginx/nginx.conf /etc/nginx/nginx.conf.bak
cp /var/www/html/forum.php /var/www/html/forum.php.bak
mkdir -p /etc/nginx/ssl
openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
  -keyout /etc/nginx/ssl/server.key -out /etc/nginx/ssl/server.crt \
  -subj "/C=CO/ST=Bogota/L=Bogota/O=AuraUniversity/CN=192.123.10.10"
# edición de /etc/nginx/nginx.conf (sección 4.3) y creación de .user.ini (sección 4.4)
nginx -t && nginx -s reload
ps aux | grep php-fpm ; kill -USR2 <PID del proceso maestro>
sed -i "s/echo \$row\['content'\];/echo htmlspecialchars(\$row['content']);/" /var/www/html/forum.php
grep -n "row\['content'\]" /var/www/html/forum.php
